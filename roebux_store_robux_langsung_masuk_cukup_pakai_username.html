2. Endpoint Webhook Pembayaran & Auto-Order (app/api/webhook/payment/route.ts) 
import { NextResponse } from 'next/server';
import { createClient } from '@supabase/supabase-js';

const supabase = createClient(process.env.NEXT_PUBLIC_SUPABASE_URL!, process.env.SUPABASE_SERVICE_ROLE_KEY!);

export async function POST(req: Request) {
  const payload = await req.json();
  const invoiceNumber = payload.merchant_ref || payload.order_id;
  const status = payload.status || payload.transaction_status;

  if (status === 'PAID' || status === 'settlement') {
    const { data: tx } = await supabase.from('transactions').select('*').eq('invoice_number', invoiceNumber).single();
    if (!tx || tx.order_status === 'SUCCESS') return NextResponse.json({ message: 'Already processed' });

    await supabase.from('transactions').update({ payment_status: 'PAID', order_status: 'PROCESSING' }).eq('invoice_number', invoiceNumber);
    const sendResult = await executeRobuxAutoSend({ robloxUserId: tx.roblox_user_id, robuxAmount: tx.robux_amount });

    await supabase.from('transactions').update({
      order_status: 'SUCCESS',
      auto_order_response: sendResult,
      completed_at: new Date().toISOString()
    }).eq('invoice_number', invoiceNumber);
  }
  return NextResponse.json({ received: true });
}

// ============================================================================
// OPSI A: GENERATE QRIS DINAMIS VIA API PAYMENT GATEWAY (MIDTRANS / TRIPAY)
// Kelebihan: Otomatis dapat Webhook LUNAS tanpa cek mutasi manual
// ============================================================================
export async function createGatewayDynamicQris(invoice: string, amountIdr: number) {
  const res = await fetch('https://api.midtrans.com/v2/charge', {
    method: 'POST',
    headers: {
      'Accept': 'application/json',
      'Content-Type': 'application/json',
      'Authorization': 'Basic ' + Buffer.from(process.env.MIDTRANS_SERVER_KEY + ':').toString('base64')
    },
    body: JSON.stringify({
      payment_type: 'qris',
      transaction_details: { order_id: invoice, gross_amount: amountIdr },
      qris: { acquirer: 'gopay' }
    })
  });
  return await res.json(); // Mengembalikan actions[0].url (Gambar QRIS) & qr_string dinamis
}

// ============================================================================
// OPSI B: UBAH QRIS STATIS PRIBADI JADI QRIS DINAMIS (STANDAR EMVCo + CRC16)
// Kelebihan: Tanpa daftar Payment Gateway, nominal otomatis muncul saat di-scan
// ============================================================================
// String QRIS Statis Asli ROEBUX - TELECOM (NMID: ID1025464550819, A01, BCA 93600014)
export const ROEBUX_STATIC_QRIS =
  '00020101021126650013ID.CO.BCA.WWW011893600014000406775102150008850040677510303UMI51440014ID.CO.QRIS.WWW0215ID10254645508190303UMI5204481453033605802ID5906ROEBUX6006BEKASI61051713562070703A0163040085';

export function convertStaticToDynamicQris(staticQris: string = ROEBUX_STATIC_QRIS, amountIdr: number): string {
  // 1. Hapus 4 karakter checksum CRC16 lama di ujung string (setelah "6304")
  const withoutCrc = staticQris.slice(0, -4);
  // 2. Ubah kode penanda statis "010211" menjadi dinamis "010212"
  const step1 = withoutCrc.replace('010211', '010212');
  // 3. Buat Tag 54 (Transaction Amount): "54" + 2 digit panjang nominal + nominal
  const amountStr = String(Math.floor(amountIdr));
  const tag54 = '54' + String(amountStr.length).padStart(2, '0') + amountStr;
  // 4. Sisipkan Tag 54 tepat sebelum kode negara "5802ID"
  const parts = step1.split('5802ID');
  const payloadToHash = parts[0] + tag54 + '5802ID' + parts[1];

  // 5. Hitung ulang Checksum CRC16-CCITT (Polynomial 0x1021, Init 0xFFFF)
  let crc = 0xFFFF;
  for (let i = 0; i < payloadToHash.length; i++) {
    crc ^= payloadToHash.charCodeAt(i) << 8;
    for (let j = 0; j < 8; j++) {
      crc = (crc & 0x8000) ? ((crc << 1) ^ 0x1021) : (crc << 1);
    }
  }
  const crcHex = (crc & 0xFFFF).toString(16).toUpperCase().padStart(4, '0');
  return payloadToHash + crcHex;
}

5. Roblox Username & Avatar Checker (netlify/functions/roblox-check.js) 
// Netlify Function: cek username Roblox (proxy, karena API Roblox tidak bisa dipanggil langsung dari browser/CORS)
// Taruh di: netlify/functions/roblox-check.js  (Node 18+)
// Endpoint: /.netlify/functions/roblox-check?u=USERNAME

const headers = {
  "Content-Type": "application/json",
  "Cache-Control": "no-store",
};

exports.handler = async (event) => {
  const u = ((event.queryStringParameters || {}).u || "").trim();
  if (!/^[A-Za-z0-9_]{3,20}$/.test(u)) {
    return { statusCode: 400, headers, body: JSON.stringify({ found: false, error: "invalid" }) };
  }

  try {
    const r = await fetch("https://users.roblox.com/v1/usernames/users", {
      method: "POST",
      headers: { "Content-Type": "application/json" },
      body: JSON.stringify({ usernames: [u], excludeBannedUsers: true }),
    });
    if (!r.ok) throw new Error("roblox " + r.status);
    const user = ((await r.json()).data || [])[0];
    if (!user) return { statusCode: 200, headers, body: JSON.stringify({ found: false }) };

    let avatar = "";
    try {
      const t = await fetch(
        "https://thumbnails.roblox.com/v1/users/avatar-headshot?userIds=" +
          user.id + "&size=150x150&format=Png&isCircular=true"
      );
      avatar = (((await t.json()).data || [])[0] || {}).imageUrl || "";
    } catch (_) {}

    return {
      statusCode: 200,
      headers,
      body: JSON.stringify({
        found: true,
        id: user.id,
        name: user.name,
        displayName: user.displayName,
        avatar,
      }),
    };
  } catch (e) {
    return { statusCode: 502, headers, body: JSON.stringify({ found: false, error: "upstream" }) };
  }
};
